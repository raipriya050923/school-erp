Program.cs ============================================================

using System.Net; using System.Net.Http.Headers; using System.Text;
using System.Text.Json;

const string tenantId = “xxx”; const
string clientId = “xxx”;

const string fromEmail = “noreply@kssinfonet.com”; const string toEmail
= “kssinfonet@gmail.com”;

var clientSecret =
Environment.GetEnvironmentVariable(“KSS_MAIL_CLIENT_SECRET”);

if (string.IsNullOrWhiteSpace(clientSecret)) { Console.WriteLine(“ERROR:
KSS_MAIL_CLIENT_SECRET environment variable not found.”); return; }

Console.WriteLine($“Client Secret loaded securely. Length:
{clientSecret.Length}”);

using var httpClient = new HttpClient();

try { // STEP 1: Get OAuth 2.0 access token

    Console.WriteLine("Getting Microsoft Graph access token...");

    var tokenUrl =
        $"https://login.microsoftonline.com/{tenantId}/oauth2/v2.0/token";

    using var tokenContent = new FormUrlEncodedContent(
        new Dictionary<string, string>
        {
            ["client_id"] = clientId,
            ["client_secret"] = clientSecret,
            ["scope"] = "https://graph.microsoft.com/.default",
            ["grant_type"] = "client_credentials"
        });

    using var tokenResponse =
        await httpClient.PostAsync(tokenUrl, tokenContent);

    var tokenResponseText =
        await tokenResponse.Content.ReadAsStringAsync();

    if (!tokenResponse.IsSuccessStatusCode)
    {
        Console.WriteLine();
        Console.WriteLine("TOKEN ERROR:");
        Console.WriteLine(tokenResponseText);
        return;
    }

    using var tokenJson =
        JsonDocument.Parse(tokenResponseText);

    var accessToken =
        tokenJson.RootElement
            .GetProperty("access_token")
            .GetString();

    if (string.IsNullOrWhiteSpace(accessToken))
    {
        Console.WriteLine("ERROR: Access token was not returned.");
        return;
    }

    Console.WriteLine("Access token received successfully.");

    // STEP 2: Prepare test email

    var mailData = new
    {
        message = new
        {
            subject = "KSS INFONET - .NET Graph Mail Test",

            body = new
            {
                contentType = "HTML",

                content = """
                    <h2>KSS INFONET</h2>
                    <p>
                        This email was sent from the
                        <strong>.NET test application</strong>.
                    </p>
                    <p>
                        Microsoft Graph OAuth 2.0 and Exchange
                        Application RBAC are working successfully.
                    </p>
                    <p>
                        From:
                        <strong>noreply@kssinfonet.com</strong>
                    </p>
                    """
            },

            toRecipients = new[]
            {
                new
                {
                    emailAddress = new
                    {
                        address = toEmail
                    }
                }
            }
        },

        saveToSentItems = true
    };

    var mailJson = JsonSerializer.Serialize(mailData);

    // STEP 3: Send using Microsoft Graph

    Console.WriteLine($"Sending email to {toEmail}...");

    var sendMailUrl =
        $"https://graph.microsoft.com/v1.0/users/{fromEmail}/sendMail";

    using var request =
        new HttpRequestMessage(HttpMethod.Post, sendMailUrl);

    request.Headers.Authorization =
        new AuthenticationHeaderValue("Bearer", accessToken);

    request.Content =
        new StringContent(
            mailJson,
            Encoding.UTF8,
            "application/json");

    using var mailResponse =
        await httpClient.SendAsync(request);

    var responseText =
        await mailResponse.Content.ReadAsStringAsync();

    if (mailResponse.StatusCode == HttpStatusCode.Accepted)
    {
        Console.WriteLine();
        Console.WriteLine("==============================");
        Console.WriteLine("EMAIL SENT SUCCESSFULLY");
        Console.WriteLine("HTTP Status: 202 Accepted");
        Console.WriteLine($"From: {fromEmail}");
        Console.WriteLine($"To:   {toEmail}");
        Console.WriteLine("==============================");
    }
    else
    {
        Console.WriteLine();
        Console.WriteLine("EMAIL SEND FAILED");
        Console.WriteLine(
            $"HTTP Status: {(int)mailResponse.StatusCode} " +
            $"{mailResponse.StatusCode}");

        Console.WriteLine(responseText);
    }

} catch (Exception ex) { Console.WriteLine();
Console.WriteLine(“UNEXPECTED ERROR:”); Console.WriteLine(ex.Message); }
